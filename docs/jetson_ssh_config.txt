# ~/.ssh/config 模板 —— SSH 到 Orin 板子（见 docs/JETSON_SSH.md）
#
# 用法：
#   mkdir -p ~/.ssh && chmod 700 ~/.ssh
#   cp docs/jetson_ssh_config.txt ~/.ssh/config
#   chmod 600 ~/.ssh/config
#
# 前提：一把权限 600 的私钥已在 WSL 内；IdentityFile 的现役文件名以 owner 机
#       ~/.ssh/config 为准（本模板初版写的 board_ed25519 已不存在）。
#       ⚠️ 不要指向 /mnt/c 下的私钥——/mnt/c 权限是 777，SSH 会以
#       "permissions too open" 拒绝加载。

Host jetson
    HostName <host-ip>
    User jetson
    IdentityFile ~/.ssh/<identity-file>
    StrictHostKeyChecking accept-new
